cmake_minimum_required(VERSION 3.24)
project(EnergyMechanismHost VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core Gui Widgets Bluetooth SerialPort Test)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(em_wire STATIC ../Common/Src/debug_protocol.c ../Common/Src/comm_v2.c)
target_include_directories(em_wire PUBLIC ../Common/Inc)
add_library(em_client STATIC src/protocol.cpp src/transport.cpp src/client.cpp)
target_include_directories(em_client PUBLIC src)
target_link_libraries(em_client PUBLIC em_wire Qt6::Core Qt6::Bluetooth Qt6::SerialPort)
add_library(em_ui STATIC src/window.cpp src/plots.cpp)
target_link_libraries(em_ui PUBLIC em_client Qt6::Widgets)
qt_add_executable(EnergyMechanismHost src/main.cpp)
target_link_libraries(EnergyMechanismHost PRIVATE em_ui)
set_target_properties(EnergyMechanismHost PROPERTIES WIN32_EXECUTABLE TRUE)
if(UNIX)
  set_target_properties(EnergyMechanismHost PROPERTIES INSTALL_RPATH "$ORIGIN/lib")
endif()
add_custom_command(TARGET EnergyMechanismHost POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/resources" "$<TARGET_FILE_DIR:EnergyMechanismHost>/resources")
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(host_tests tests/test_host.cpp)
  target_link_libraries(host_tests PRIVATE em_ui Qt6::Test)
  target_compile_definitions(host_tests PRIVATE EM_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME host_protocol_session COMMAND host_tests)
  set_tests_properties(host_protocol_session PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME host_ui_smoke COMMAND EnergyMechanismHost --smoke-test "${CMAKE_CURRENT_BINARY_DIR}/screenshots")
  set_tests_properties(host_ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(usb_gateway_tests ../ESP32/src/gateway_core.c ../ESP32/src/usb_gateway.c ../tests/test_usb_gateway.c)
  target_include_directories(usb_gateway_tests PRIVATE ../ESP32/src)
  target_link_libraries(usb_gateway_tests PRIVATE em_wire)
  if(MSVC)
    target_compile_options(usb_gateway_tests PRIVATE /UNDEBUG)
  else()
    target_compile_options(usb_gateway_tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME usb_four_clients COMMAND usb_gateway_tests)
endif()
install(TARGETS EnergyMechanismHost RUNTIME DESTINATION .)
install(FILES README.md DESTINATION .)
install(DIRECTORY resources/ DESTINATION resources)
